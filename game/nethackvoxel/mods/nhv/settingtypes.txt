# Draw monsters, items and features as flat sprites instead of extruded
# voxel models. (Also in the game: Inventory > Graphics.)
nhv_flat_sprites (Flat sprites) bool false

# Monsters, objects and you cast hard shadows: your sprites, blacked out, on the floor
nhv_shadows (Shadows) bool true

# How dark shadows are, in percent
nhv_shadow_opacity (Shadow opacity) int 60 0 100

# The shadows drawn on the monster and object pictures (off: a copy without them; after a restart)
nhv_sprite_shadows (Shadows drawn on sprites) bool true

# Interface style: retro (NetHack's IBM PC pixel font, old PC style boxes)
# or modern (Luanti's own look). (Also in the game: Inventory > Graphics.)
nhv_interface (Interface style) enum retro retro,modern

# Pixels per map square of the corner map (0 hides it).
nhv_map_scale (Map size) int 3 0 8

# The big map: you and the monsters drawn where they really are, not on
# their squares. (Also in the game: Options > Map icons.)
nhv_map_exact (Map icons where they really are) bool false

# Text size (font size multiplier, in menus and on the HUD). The retro
# pixel font only uses whole steps (1, 2, 3, 4: 14, 28, 42, 56 pixels,
# whatever the HUD size).
nhv_text_scale (Text size) float 1 0.5 4

# Check for a newer VoxHack release when the game starts (it asks GitHub
# once a session; the title screen says if there is one).
nhv_update_check (Check for updates) bool true
