# nhcore: NetHack 3.7 built as a self-contained DLL (nethack.dll) with a
# small C API (nhv_api.h) that the Luanti engine drives.
#
# Lua 5.4 (needed by NetHack's level scripts) is linked statically into the
# DLL and not exported, so it cannot clash with Luanti's LuaJIT.
cmake_minimum_required(VERSION 3.20)
project(nhcore C CXX)

set(NH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../external/nethack" CACHE PATH "NetHack source tree")
get_filename_component(NH_ROOT "${NH_ROOT}" ABSOLUTE)
set(LUA_DIR "${NH_ROOT}/lib/lua-5.4.8/src")
set(GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/gen")
set(NHV_DATA_OUT "${CMAKE_CURRENT_BINARY_DIR}/nhdata" CACHE PATH "Where runtime data files are assembled")

if(NOT EXISTS "${LUA_DIR}/lua.h")
    message(FATAL_ERROR "Lua 5.4 source missing. Run: sh sys/windows/fetch.sh lua (in ${NH_ROOT})")
endif()

set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)

# ---------------------------------------------------------------- generated
file(MAKE_DIRECTORY "${GEN_DIR}")
file(WRITE "${GEN_DIR}/nhlua.h.in"
"/* nhlua.h - generated by nhcore/CMakeLists.txt */
#include \"lua.h\"
ATTRNORETURN LUA_API int (lua_error) (lua_State *L) NORETURN;
#include \"lualib.h\"
#include \"lauxlib.h\"
/*nhlua.h*/
")
configure_file("${GEN_DIR}/nhlua.h.in" "${GEN_DIR}/nhlua.h" COPYONLY)

set(NH_COMMON_DEFS WIN32 WINVER=0x0601 _WIN32_WINNT=0x0601 _LIB NO_TILE_C)
set(NH_COMMON_INCS "${GEN_DIR}" "${NH_ROOT}/include" "${NH_ROOT}/sys/windows" "${LUA_DIR}")
set(NH_WARN -Wno-unused-parameter -Wno-deprecated-declarations
    -Wno-missing-field-initializers -Wno-unused-variable)

# ---------------------------------------------------------------- lua 5.4
file(GLOB LUA_SRCS "${LUA_DIR}/*.c")
list(REMOVE_ITEM LUA_SRCS "${LUA_DIR}/lua.c" "${LUA_DIR}/luac.c" "${LUA_DIR}/onelua.c")
add_library(nhlua54 STATIC ${LUA_SRCS})
target_include_directories(nhlua54 PUBLIC "${LUA_DIR}")
set_target_properties(nhlua54 PROPERTIES POSITION_INDEPENDENT_CODE ON)

# ---------------------------------------------------------------- makedefs
add_executable(makedefs
    "${NH_ROOT}/util/makedefs.c" "${NH_ROOT}/src/alloc.c" "${NH_ROOT}/src/date.c"
    "${NH_ROOT}/src/monst.c" "${NH_ROOT}/src/objects.c" "${NH_ROOT}/util/panic.c"
    "${NH_ROOT}/src/hacklib.c")
target_include_directories(makedefs PRIVATE ${NH_COMMON_INCS})
target_compile_definitions(makedefs PRIVATE ${NH_COMMON_DEFS})
target_compile_options(makedefs PRIVATE ${NH_WARN})

# makedefs reads/writes relative to the NetHack "dat" dir; run it in a
# private copy so the source tree stays clean.
set(DAT_WORK "${CMAKE_CURRENT_BINARY_DIR}/datwork")
set(MD_OUTPUTS data oracles options rumors bogusmon engrave epitaph)
list(TRANSFORM MD_OUTPUTS PREPEND "${DAT_WORK}/dat/" OUTPUT_VARIABLE MD_OUTPUT_FILES)
add_custom_command(
    OUTPUT ${MD_OUTPUT_FILES}
    COMMAND ${CMAKE_COMMAND} -E rm -rf "${DAT_WORK}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${DAT_WORK}/util" "${DAT_WORK}/include"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${NH_ROOT}/dat" "${DAT_WORK}/dat"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${NH_ROOT}/include" "${DAT_WORK}/include"
    COMMAND ${CMAKE_COMMAND} -E chdir "${DAT_WORK}/util" $<TARGET_FILE:makedefs> -d
    COMMAND ${CMAKE_COMMAND} -E chdir "${DAT_WORK}/util" $<TARGET_FILE:makedefs> -h
    COMMAND ${CMAKE_COMMAND} -E chdir "${DAT_WORK}/util" $<TARGET_FILE:makedefs> -v
    COMMAND ${CMAKE_COMMAND} -E chdir "${DAT_WORK}/util" $<TARGET_FILE:makedefs> -r
    COMMAND ${CMAKE_COMMAND} -E chdir "${DAT_WORK}/util" $<TARGET_FILE:makedefs> -s
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS makedefs
    COMMENT "Generating NetHack data files with makedefs")
add_custom_target(nhdat_gen DEPENDS ${MD_OUTPUT_FILES})

# ---------------------------------------------------------------- tilemap
# Generates tile.c, which fills glyphmap[].tileidx so every glyph knows its
# tile in the vanilla 16x16 tileset (win/share/*.txt).
add_executable(tilemap
    "${NH_ROOT}/win/share/tilemap.c" "${NH_ROOT}/src/drawing.c" "${NH_ROOT}/src/monst.c"
    "${NH_ROOT}/src/objects.c" "${NH_ROOT}/src/hacklib.c")
target_include_directories(tilemap PRIVATE ${NH_COMMON_INCS})
target_compile_definitions(tilemap PRIVATE ${NH_COMMON_DEFS})
target_compile_options(tilemap PRIVATE ${NH_WARN})
set(TM_WORK "${CMAKE_CURRENT_BINARY_DIR}/tmwork")
add_custom_command(
    OUTPUT "${TM_WORK}/src/tile.c"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${TM_WORK}/src" "${TM_WORK}/util"
    COMMAND ${CMAKE_COMMAND} -E chdir "${TM_WORK}/util" $<TARGET_FILE:tilemap>
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS tilemap
    COMMENT "Generating tile.c with tilemap")

# ---------------------------------------------------------------- dlb archive
add_executable(dlb
    "${NH_ROOT}/util/dlb_main.c" "${NH_ROOT}/src/dlb.c" "${NH_ROOT}/src/alloc.c"
    "${NH_ROOT}/util/panic.c" "${NH_ROOT}/src/hacklib.c")
target_include_directories(dlb PRIVATE ${NH_COMMON_INCS})
target_compile_definitions(dlb PRIVATE ${NH_COMMON_DEFS} DLB)
target_compile_options(dlb PRIVATE ${NH_WARN})

set(NHDAT_FILE "${NHV_DATA_OUT}/nhdat370")
add_custom_command(
    OUTPUT "${NHDAT_FILE}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${NHV_DATA_OUT}"
    COMMAND ${CMAKE_COMMAND} -E copy "${NH_ROOT}/sys/windows/porthelp" "${DAT_WORK}/dat/porthelp"
    COMMAND ${CMAKE_COMMAND} -DDLB=$<TARGET_FILE:dlb> -DDAT=${DAT_WORK}/dat
            -DOUT=${NHV_DATA_OUT}/nhdat -P "${CMAKE_CURRENT_SOURCE_DIR}/make_nhdat.cmake"
    DEPENDS dlb ${MD_OUTPUT_FILES} "${CMAKE_CURRENT_SOURCE_DIR}/make_nhdat.cmake"
    COMMENT "Building nhdat370 archive")

# Runtime data directory: the archive plus the few loose files NetHack
# reads directly.
add_custom_target(nhdata ALL
    COMMAND ${CMAKE_COMMAND} -E copy "${NH_ROOT}/sys/windows/sysconf.template" "${NHV_DATA_OUT}/sysconf"
    COMMAND ${CMAKE_COMMAND} -E copy "${NH_ROOT}/dat/symbols" "${NHV_DATA_OUT}/symbols"
    DEPENDS "${NHDAT_FILE}")

# ---------------------------------------------------------------- nethack.dll
set(NH_CORE_NAMES allmain alloc apply artifact attrib ball bones botl
    calendar cfgfiles cmd coloratt date dbridge decl detect dig display dlb do
    do_name do_wear dog dogmove dokick dothrow drawing dungeon earlyarg eat end
    engrave exper explode extralev files fountain getpos glyphs hack hacklib
    iactions insight invent isaac64 light lock mail makemon mcastu mdlib mhitm
    mhitu minion mklev mkmap mkmaze mkobj mkroom mon mondata monmove monst
    mplayer mthrowu muse music nhlobj nhlsel nhlua nhmd4 o_init objects objnam
    options pager pickup pline polyself potion pray priest quest questpgr read
    rect region report restore rip rnd role rumors save selvar sfbase sfstruct
    shk shknam sit sounds sp_lev spell stairs steal steed strutil symbols sys
    teleport timeout topten track trap u_init uhitm utf8map vault version vision
    weapon were wield windows wizard wizcmds worm worn write zap)
list(TRANSFORM NH_CORE_NAMES PREPEND "${NH_ROOT}/src/" OUTPUT_VARIABLE NH_CORE_SRCS)
list(TRANSFORM NH_CORE_SRCS APPEND ".c")

add_library(nethack SHARED
    ${NH_CORE_SRCS}
    "${NH_ROOT}/sys/share/random.c"
    "${NH_ROOT}/sys/share/cppregex.cpp"
    "${NH_ROOT}/sys/windows/windsys.c"
    "${TM_WORK}/src/tile.c"
    nhv_main.c
    nhv_win.c
    nhv_api.c
    nhv_free.c
    nhv_sound.c)
target_include_directories(nethack PRIVATE ${NH_COMMON_INCS} "${CMAKE_CURRENT_SOURCE_DIR}")
list(REMOVE_ITEM NH_COMMON_DEFS NO_TILE_C)
target_compile_definitions(nethack PRIVATE ${NH_COMMON_DEFS} DLB NHV_FREEMOVE SND_LIB_VISSOUND SND_SPEECH TILES_IN_GLYPHMAP SHIM_GRAPHICS NOTTYGRAPHICS NHV_BUILDING_DLL)
target_compile_options(nethack PRIVATE ${NH_WARN} $<$<COMPILE_LANGUAGE:C>:-include process.h>)
target_link_libraries(nethack PRIVATE nhlua54 shlwapi ole32 shell32 uuid winmm bcrypt userenv rpcrt4)
set_target_properties(nethack PROPERTIES PREFIX "")
# no runtime DLL dependencies besides Windows itself
target_link_options(nethack PRIVATE -static)
# windsys.c defines these too; ours (in nhv_main.c) take precedence.
set_source_files_properties("${NH_ROOT}/sys/windows/windsys.c" PROPERTIES COMPILE_DEFINITIONS
    "nethack_exit=windsys_nethack_exit_unused;set_default_prefix_locations=windsys_set_default_prefix_locations_unused")

# ---------------------------------------------------------------- test harness
add_executable(nhvtest test/nhvtest.c)
target_include_directories(nhvtest PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(nhvtest PRIVATE nethack)
target_link_options(nhvtest PRIVATE -static)
